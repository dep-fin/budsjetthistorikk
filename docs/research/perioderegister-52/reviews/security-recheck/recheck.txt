RESULTS: list[dict] = []
CALLS: list[str] = []


def digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def git(*args: str) -> str:
    return subprocess.check_output(["git", *args], text=True).strip()


def record(name: str, status: str, **details) -> None:
    if any(r["name"] == name for r in RESULTS):
        return
    row = dict(name=name, status=status, elapsed=round(time.monotonic() - START, 3), **details)
    RESULTS.append(row)
    with (OUT / "results.jsonl").open("a", encoding="utf8") as stream:
        stream.write(json.dumps(row, ensure_ascii=False) + "\n")
    print(json.dumps(row, ensure_ascii=True), flush=True)


def forbidden(*args, **kwargs):
    CALLS.append(repr(args))
    raise AssertionError("Network and SQLite are forbidden in this local audit")


def preserve() -> dict[str, str]:
    paths = set((ROOT / "docs/research/perioderegister-52").rglob("*"))
    paths.update((OLD / "inputs/p1-complete").glob("*"))
    paths.update(OLD / "synthetic" / name for name in (
        "container-root-external-office-document.xlsx", "container-apply-alignment-zero.xlsx"))
    paths.update(ROOT / name for name in git("ls-files").splitlines()
                 if name.endswith((".py", ".mjs")))
    paths.update(OLD / name for name in (
        "candidate.json", "artifact-manifest.json", "confirmed-findings.json", "REPORT.md"))
    return {p.relative_to(ROOT).as_posix(): digest(p) for p in sorted(paths) if p.is_file()}


def copy_input(source: Path) -> Path:
    target = OUT / "inputs" / source.parent.name / source.name
    target.parent.mkdir(parents=True, exist_ok=True)
    if not target.exists():
        shutil.copyfile(source, target)
    assert digest(source) == digest(target)
    return target


def rewrite(source: Path, name: str, changes: dict, extra: dict | None = None) -> Path:
    target = OUT / "synthetic" / (name + ".xlsx")
    completed = next((r for r in RESULTS if r["name"] == name), None)
    if completed:
        target = OUT / completed["file"]
        assert digest(target) == completed["sha256"]
        return target
    if target.exists():
        target = target.with_stem(target.stem + "-retry1")
    assert not target.exists()
    with ZipFile(source) as old, ZipFile(target, "w", ZIP_DEFLATED) as new:
        for info in old.infolist():
            raw = old.read(info.filename)
            if info.filename in changes:
                root = etree.fromstring(raw)
                changes[info.filename](root)
                raw = etree.tostring(root)
            new.writestr(info, raw)
        for member, raw in (extra or {}).items():
            new.writestr(member, raw)
    return target


def rejected(name: str, path: Path) -> None:
    if any(r["name"] == name for r in RESULTS):
        return
    try:
        report = excel.read_excel(path)
    except ValueError as error:
        record(name, "REJECTED", error=str(error), file=path.relative_to(OUT).as_posix(), sha256=digest(path))
    except Exception as error:
        record(name, "UNEXPECTED_EXCEPTION", error=repr(error), file=path.relative_to(OUT).as_posix(), sha256=digest(path))
    else:
        record(name, "ACCEPTED", report_id=report.id, file=path.relative_to(OUT).as_posix(), sha256=digest(path))


def findrel(root, suffix: str):
    return next(e for e in root if e.get("Type", "").endswith("/" + suffix))


def setrel(suffix: str, attr: str, value: str):
    return lambda r: findrel(r, suffix).set(attr, value)


def duplicate(root):
    root.append(copy.deepcopy(root[0]))


def missing_id(root):
    del root[0].attrib["Id"]


def duplicate_id(root):
    root[1].set("Id", root[0].get("Id"))


def append_external(root):
    e = etree.SubElement(root, P + "Relationship")
    e.attrib.update({"Id": "unused", "Type": "http://schemas.openxmlformats.org/officeDocument/2006/relationships/hyperlink",
                     "Target": "https://example.invalid/local-test", "TargetMode": "External"})


def change_type(part: str, value: str):
    return lambda r: next(e for e in r if e.get("PartName") == part).set("ContentType", value)


def raw_style(group: str, index: int, attr: str, value: str | None):
    def change(root):
        element = root.find(S + group)[index]
        if value is None:
            element.attrib.pop(attr, None)
        else:
            element.set(attr, value)
    return change

                   global_counts=global_counts, json_sha256=digest(frozen), xlsx_sha256=digest(workbook))
        assert [c.amount_kr for c in reference.calculations] == [166039000, 29185000]
