START = time.monotonic()

def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def record(name, status, **details):
    row = dict(name=name, status=status, seconds=round(time.monotonic()-START, 3), **details)
    with LOG.open('a', encoding='utf8') as stream:
        stream.write(json.dumps(row, ensure_ascii=False) + '\n')
    print(json.dumps({k:v for k,v in row.items() if k in ('name','status','seconds','error')}, ensure_ascii=True), flush=True)

def copied(source):
    target = OUT / 'inputs' / source.parent.name / source.name
    target.parent.mkdir(parents=True, exist_ok=True)
    before = digest(source)
    shutil.copyfile(source, target)
    assert digest(target) == before == digest(source)
    return target

def rejected(name, callback, **details):
    try:
        callback()
    except ValueError as error:
        record(name, 'rejected', error=str(error), **details)
    except Exception as error:
        record(name, 'unexpected_exception', error=repr(error), **details)
    else:
        record(name, 'ACCEPTED', **details)

def rehash(data):
    """Reidentify every P1 object and all references in topological order."""
    report = data['report']; basis = report['basis']
    pairs = [('sources','periodekilde-1'), ('evidence','periodebevis-1'),
             ('documents','periodedokument-1'), ('inventory','periodeinventar-1'),
             ('statements','periodeutsagn-1'), ('relations','perioderelasjon-1'),
             ('entries','periodeledd-1'), ('case_candidates','periodesakskandidat-1'),
             ('decision_candidates','periodevedtakskandidat-1'), ('gaps','periodehull-1')]
    mapping = {}
    def update(obj):
        if isinstance(obj, str): return mapping.get(obj, obj)
        if isinstance(obj, list): return [update(x) for x in obj]
        if isinstance(obj, dict): return {k:(v if k == 't4_basis' else update(v)) for k,v in obj.items()}
        return obj
    for collection, namespace in pairs:
        basis[collection] = update(basis[collection])
        for obj in basis[collection]:
            old = obj['id']; obj['id'] = ''
            obj['id'] = identity(namespace, obj)
            mapping[old] = obj['id']
    report = update(report)
    report['basis']['id'] = ''
    report['basis']['id'] = identity('periodegrunnlag-1', report['basis'])
    report['id'] = ''
    report['id'] = identity('perioderegister-1', report)
    data['report'] = report
    return encode(data)

def zip_patch(source, target, member, change, extra=None):
    with ZipFile(source) as old, ZipFile(target, 'w', ZIP_DEFLATED) as new:
        for info in old.infolist():
            raw = old.read(info.filename)
            if info.filename == member: raw = change(raw)
            new.writestr(info, raw)
        if extra: new.writestr(*extra)

def xml_change(raw, change):
    tree = etree.fromstring(raw)
    change(tree)
    return etree.tostring(tree)

    record('positive-navigation-and-old-calculations','passed')

def finish(report, inputs, source, code):
    # Additional display/container variants are generated from a copied valid book.
    probes={
      'white-font':('xl/styles.xml',lambda t:t.find(S+'fonts')[1].find(S+'color').set('rgb','00FFFFFF')),
      'apply-alignment-zero':('xl/styles.xml',lambda t:[x.set('applyAlignment','0') for x in t.find(S+'cellXfs')]),
      'root-external-office-document':('_rels/.rels',lambda t:(t[0].set('Target','https://example.invalid/synthetic.xlsx'),t[0].set('TargetMode','External'))),
      'wrong-visible-value':('xl/worksheets/sheet1.xml',lambda t:t.find('.//'+S+'c[@r="B2"]/'+S+'v').__setattr__('text','999999999')),
      'hidden-row':('xl/worksheets/sheet1.xml',lambda t:t.find(S+'sheetData/'+S+'row[@r="2"]').set('hidden','1')),
      'formula':('xl/worksheets/sheet1.xml',lambda t:etree.SubElement(t.find('.//'+S+'c[@r="G2"]'),S+'f').__setattr__('text','1+1')),
    }
    for name,(member,change) in probes.items():
        if '--resume' in sys.argv and name in {'white-font','apply-alignment-zero','root-external-office-document'}:
            continue
        x=OUT/'synthetic'/('container-'+name+'.xlsx')
        zip_patch(inputs['both'][1],x,member,lambda raw:xml_change(raw,change))
        rejected('xlsx-'+name,lambda c=x:excel.read_excel(c),sha256=digest(x))
    for source_name in ('duplicate-member','traversal','external-relationship','huge-dimension','member-limit','doctype','expanded-limit'):
        x=copied(ROOT/'.local/p1-guards-complete'/(source_name+'.xlsx'))
        rejected('copied-container-'+source_name,lambda c=x:excel.read_excel(c),sha256=digest(x))
    # Publication must never replace an existing output even when the target appears at publish time.
    for kind,exporter in [('json',p1.export_json),('xlsx',excel.export_excel)]:
        target=OUT/'synthetic'/('existing.'+kind); target.write_bytes(b'preserve-existing-sentinel')
        rejected('publish-preserves-existing-'+kind,lambda e=exporter,t=target:e(report,t))
        assert target.read_bytes()==b'preserve-existing-sentinel'
    finish_ordinary(report, inputs, source, code)

    from ops.visningspilot_proeve import _ordinary
    from innlesning.registers import seed_register,read_register

    from innlesning import read_extraction
    ordinary=OUT/'ordinary-368'; ordinary.mkdir(exist_ok=True)
    assert not (ordinary/'ordinary.sqlite').exists(), 'Do not reuse an existing database'
    begin_source_attempt(ordinary/'ordinary.sqlite','prove','2024-2025')
    register=seed_register(ordinary/'ordinary.sqlite')
    stats=_ordinary(ordinary)
    assert stats['observations']==222 and stats['events']==74
    assert read_register(ordinary/'ordinary.sqlite',register.id)==register
    assert {o.round for o in read_extraction(ordinary/'ordinary.sqlite',str(stats['extraction_id'])).observations}=={'uavklart'}
    with sqlite3.connect(ordinary/'ordinary.sqlite') as c: assert c.execute('PRAGMA user_version').fetchone()[0]==5
    record('ordinary-368-new-public-cli-db-raw-extraction-excel-b5-schema5','passed',details=stats)
    # Public P1 CLI with inaccessible originals, no socket/SQLite, empty working directory.
    offline=OUT/'offline'; offline.mkdir()
    env=dict(os.environ); env['PYTHONPATH']=os.pathsep.join([str(ROOT),str(ROOT/'.local/python')])
    script="""import socket, sqlite3, sys
def forbidden(*a,**kw): raise AssertionError('audit offline: network/SQLite forbidden')
socket.create_connection=forbidden
socket.socket.connect=forbidden
socket.socket.connect_ex=forbidden
sqlite3.connect=forbidden
sys.argv=['analyse.perioderegister','--frosset',sys.argv[1],'--rot',sys.argv[2], '--json',sys.argv[3],'--excel',sys.argv[4]]
from analyse.perioderegister import main
main()
"""
    command=[sys.executable,'-c',script,str(inputs['both'][0]),str(offline/'missing-originals'),str(offline/'reexport.json'),str(offline/'reexport.xlsx')]
    result=subprocess.run(command,cwd=offline,env=env,capture_output=True,text=True,encoding='utf8',timeout=240)
    (OUT/'offline-cli.json').write_text(json.dumps(dict(returncode=result.returncode,stdout=result.stdout,stderr=result.stderr),indent=2),encoding='utf8')
    assert result.returncode==0,result.stderr
    assert p1.read_json(offline/'reexport.json')==excel.read_excel(offline/'reexport.xlsx')==report
    record('offline-public-cli-reexport-closed-files','passed',originals_absent=True,network_sqlite_blocked=True)
    assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()==HEAD
    assert not subprocess.check_output(['git','status','--porcelain=v1'],text=True).strip()
    for name,sha in code.items(): assert digest(ROOT/name)==sha
    for source_name in ('both','vedtak346','empty','wrong-year','unknown-index'):
        for suffix in ('.json','.xlsx'):
            assert digest(source/(source_name+suffix))==digest(OUT/'inputs/p1-complete'/(source_name+suffix))
    record('candidate-and-source-inputs-remain-unchanged','passed')
